const PORT = 38217;
const post = (body) => fetch(`http://127.0.0.1:${PORT}/result`, {
  method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(body)
}).catch(() => {});

chrome.runtime.onInstalled.addListener(async () => {
  await post({ event: "sw_installed", at: Date.now() });
  const tab = await chrome.tabs.create({ url: chrome.runtime.getURL("probe.html"), active: false });
  await post({ event: "probe_tab_created", tab_id: tab.id, active: tab.active, at: Date.now() });
});

// Every SW -> page message is a BROADCAST. A terminated MV3 service worker
// loses its module state, so a remembered tab id is null by the time an alarm
// wakes it. (Learned the hard way: the first probe run stalled on exactly this.)
chrome.alarms.onAlarm.addListener(async (alarm) => {
  const at = Date.now();
  if (alarm.name === "begin") {
    await post({ event: "sw_alarm_begin", at });
    chrome.runtime.sendMessage({ type: "begin" }).catch(() => {});
    return;
  }
  if (alarm.name.startsWith("swalarm:")) {
    const started = Number(alarm.name.split(":")[1]);
    chrome.runtime.sendMessage({ type: "sw_alarm_done", elapsed_ms: at - started }).catch(() => {});
  }
});

chrome.runtime.onMessage.addListener((msg, sender, respond) => {
  if (msg?.type === "hide_me") {
    // A background tab in a focused window is unambiguously hidden.
    chrome.tabs.create({ url: "about:blank", active: true }).then(() => respond({ ok: true }));
    return true;
  }
  if (msg?.type === "show_me") {
    chrome.tabs.update(sender.tab.id, { active: true }).then(() => respond({ ok: true }));
    return true;
  }
  if (msg?.type === "arm_begin") {
    // Long enough to cross Chrome's 5-minute grace period and enter intensive
    // wake-up throttling -- the regime the live 11-minute run was in.
    chrome.alarms.create("begin", { delayInMinutes: 5.5 });
    respond({ ok: true });
    return true;
  }
  if (msg?.type === "sw_alarm_start") {
    const now = Date.now();
    chrome.alarms.create(`swalarm:${now}`, { when: now + msg.ms });
    respond({ ok: true });
    return true;
  }
  if (msg?.type === "sw_settimeout_start") {
    const now = Date.now();
    setTimeout(() => {
      chrome.runtime.sendMessage({ type: "sw_settimeout_done", elapsed_ms: Date.now() - now }).catch(() => {});
    }, msg.ms);
    respond({ ok: true });
    return true;
  }
  return false;
});
