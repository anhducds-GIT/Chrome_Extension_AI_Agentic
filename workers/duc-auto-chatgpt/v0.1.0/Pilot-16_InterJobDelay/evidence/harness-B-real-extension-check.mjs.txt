// Verifies the REAL shipped module inside a REAL hidden side-panel document.
//
// This is not the live pilot (that needs chatgpt.com and Duc's approval). It is
// the step in between: it proves interjob-delay-core.js, as loaded by the real
// sidepanel.html in real Chrome with the real chrome.alarms API, waits the
// configured gap while the document is hidden -- and that the OLD tick loop
// does not. No job is run, no ChatGPT tab is opened, throwaway profile.
import { httpJson, connect } from "./cdp.mjs";
import { spawn } from "node:child_process";
import fs from "node:fs";
import path from "node:path";

const PORT = Number(process.env.PORT || 9336);
const EXT = "C:/WORKING ZONE/Chrome_Extension_AI_Agentic/workers/duc-auto-chatgpt/v0.1.0";
const HERE = path.dirname(new URL(import.meta.url).pathname.replace(/^\//, ""));
const PROFILE = path.join(HERE, "chrome-real-ext");
const CHROME = "C:/Program Files/Google/Chrome/Application/chrome.exe";
const GRACE_MS = Number(process.env.GRACE_MS || 340000);
const SECONDS = 12;

const log = (...a) => console.log(new Date().toISOString(), ...a);
const wait = (ms) => new Promise((r) => setTimeout(r, ms));

fs.rmSync(PROFILE, { recursive: true, force: true });
fs.mkdirSync(PROFILE, { recursive: true });

const child = spawn(CHROME, [
  `--user-data-dir=${PROFILE}`,
  `--remote-debugging-port=${PORT}`,
  "--remote-allow-origins=*",
  "--enable-unsafe-extension-debugging",
  "--no-first-run", "--no-default-browser-check",
  "--window-size=900,700",
  "about:blank"
], { detached: true, stdio: "ignore" });
child.unref();

await wait(6000);
const version = await httpJson(PORT, "/json/version");
log("chrome up:", version.Browser);
const browser = await connect(version.webSocketDebuggerUrl);
const { id: extId } = await browser.send("Extensions.loadUnpacked", { path: EXT });
log("real extension loaded:", extId);

// Open the real side panel document as a BACKGROUND tab. A background tab is
// document.visibilityState === "hidden", the same state a side panel is in when
// its window is not the foreground window.
const panelUrl = `chrome-extension://${extId}/sidepanel.html`;
const created = await browser.send("Target.createTarget", { url: panelUrl, background: true });
log("panel target:", created.targetId);
await wait(4000);

const targets = await httpJson(PORT, "/json/list");
const panel = targets.find((t) => t.url.startsWith(panelUrl));
if (!panel) throw new Error("side panel target not found: " + JSON.stringify(targets.map((t) => t.url)));
const page = await connect(panel.webSocketDebuggerUrl);
await page.send("Runtime.enable");

async function evaluate(expression, awaitPromise = true) {
  const res = await page.send("Runtime.evaluate", {
    expression, awaitPromise, returnByValue: true, timeout: 1200000
  });
  if (res.exceptionDetails) throw new Error(JSON.stringify(res.exceptionDetails).slice(0, 600));
  return res.result.value;
}

// 1. Is the module actually there, loaded by the real sidepanel.html?
const wiring = await evaluate(`(() => ({
  visibility: document.visibilityState,
  hasModule: typeof window.DacInterJobDelay === "object" && window.DacInterJobDelay !== null,
  api: window.DacInterJobDelay ? Object.keys(window.DacInterJobDelay).sort() : null,
  hasAlarms: typeof chrome !== "undefined" && !!chrome.alarms,
  hasRunnerCore: typeof window.DacRunnerCore === "object"
}))()`, false);
log("wiring:", JSON.stringify(wiring));
if (!wiring.hasModule) throw new Error("sidepanel.html did not load interjob-delay-core.js");
if (!wiring.hasAlarms) throw new Error("chrome.alarms is not available in the side panel");
if (wiring.visibility !== "hidden") throw new Error(`expected a hidden document, got ${wiring.visibility}`);

// 2. Install the two clocks under test, using the panel's OWN sleep shape.
await evaluate(`(() => {
  window.__probe = {
    sleep: (ms) => new Promise((r) => setTimeout(r, ms)),
    // the shape that shipped before the fix
    oldClock: async (seconds) => {
      const started = Date.now();
      for (const _ of window.DacRunnerCore.countdownValues(seconds)) await window.__probe.sleep(1000);
      return Date.now() - started;
    },
    // the shape that ships now, called exactly as sidepanel.js calls it
    newClock: async (seconds) => {
      const started = Date.now();
      const out = await window.DacInterJobDelay.waitBetweenJobs({
        seconds,
        token: "harness",
        alarms: chrome.alarms,
        sleep: window.__probe.sleep,
        shouldStop: () => false,
        onTick: () => {}
      });
      return { wall_ms: Date.now() - started, reported: out };
    }
  };
  return true;
})()`, false);

log(`staying hidden for ${Math.round(GRACE_MS / 1000)}s to cross Chrome's grace period and enter intensive throttling...`);
// Waited from Node, so the panel's own throttled timers are not involved.
await wait(GRACE_MS);
log("visibility now:", await evaluate("document.visibilityState", false));

const results = {};
log("measuring OLD clock (12s configured) in the hidden panel -- this is the bug, expect minutes");
results.old_hidden_ms = await evaluate(`window.__probe.oldClock(${SECONDS})`);
log("OLD hidden:", results.old_hidden_ms, "ms");

log("measuring NEW clock (12s configured) in the same hidden panel");
results.new_hidden = await evaluate(`window.__probe.newClock(${SECONDS})`);
log("NEW hidden:", JSON.stringify(results.new_hidden));

// 3. Control: same two clocks with the panel FOREGROUND.
await browser.send("Target.activateTarget", { targetId: created.targetId });
await wait(2500);
results.visibility_after_activate = await evaluate("document.visibilityState", false);
log("visibility now:", results.visibility_after_activate);
results.old_visible_ms = await evaluate(`window.__probe.oldClock(${SECONDS})`);
results.new_visible = await evaluate(`window.__probe.newClock(${SECONDS})`);
log("OLD visible:", results.old_visible_ms, "ms · NEW visible:", JSON.stringify(results.new_visible));

results.chrome = version.Browser;
results.extension_id = extId;
results.configured_seconds = SECONDS;
results.hidden_grace_ms = GRACE_MS;
results.wiring = wiring;
const out = path.join(HERE, "real-ext-results.json");
fs.writeFileSync(out, JSON.stringify(results, null, 2), "utf8");
log("written", out);

await browser.send("Browser.close").catch(() => {});
process.exit(0);
